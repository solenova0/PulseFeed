cmake_minimum_required(VERSION 3.20)
project(PulseFeed VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PULSEFEED_BUILD_ENGINE "Build gRPC engine (requires system protobuf/gRPC)" OFF)
option(PULSEFEED_BUILD_BENCHMARKS "Build Google Benchmark suite" OFF)
option(PULSEFEED_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(Threads REQUIRED)
if(PULSEFEED_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(orderbook OrderBook.cpp)
target_include_directories(orderbook PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(orderbook PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(orderbook PUBLIC Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE orderbook)
  add_test(NAME core_correctness COMMAND core_tests)
  add_executable(allocation_tests tests/allocation_tests.cpp)
  target_link_libraries(allocation_tests PRIVATE orderbook)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(allocation_tests PRIVATE PULSEFEED_WRAP_C_ALLOC=1)
    target_link_options(allocation_tests PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc
      -Wl,--wrap=realloc -Wl,--wrap=aligned_alloc)
  endif()
  add_test(NAME allocation_free_hot_path COMMAND allocation_tests)
endif()
if(PULSEFEED_BUILD_BENCHMARKS)
  find_package(benchmark CONFIG REQUIRED)
  add_executable(benchmarks benchmarks.cpp)
  target_link_libraries(benchmarks PRIVATE orderbook benchmark::benchmark)
endif()
if(PULSEFEED_BUILD_ENGINE)
  find_package(Protobuf REQUIRED)
  find_package(gRPC CONFIG REQUIRED)
  set(PROTO "${CMAKE_CURRENT_SOURCE_DIR}/proto/pulsefeed.proto")
  set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
  file(MAKE_DIRECTORY "${GEN}")
  add_custom_command(
    OUTPUT "${GEN}/pulsefeed.pb.cc" "${GEN}/pulsefeed.pb.h"
           "${GEN}/pulsefeed.grpc.pb.cc" "${GEN}/pulsefeed.grpc.pb.h"
    COMMAND protobuf::protoc --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto
      --cpp_out=${GEN} --grpc_out=${GEN}
      --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> "${PROTO}"
    DEPENDS "${PROTO}" protobuf::protoc gRPC::grpc_cpp_plugin VERBATIM)
  add_library(feed_proto "${GEN}/pulsefeed.pb.cc" "${GEN}/pulsefeed.grpc.pb.cc")
  target_include_directories(feed_proto PUBLIC "${GEN}")
  target_link_libraries(feed_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)
  add_executable(engine cpp/engine.cpp)
  target_link_libraries(engine PRIVATE orderbook feed_proto)
  target_compile_options(engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
